# Resultado da sonda Profit na retomada de 26/10/2026

PR #2. Controles de referência publicados em `f51b1fe289a38a347b5e542aab72e3db57099f03`. Motor do simulador `0d2f19a47307d5390f5d89bb53d70e6c87a37856`. A versão exata compilada no Profit e sua configuração completa não vieram anexadas. [Sonda e protocolo de referência](../WINZ26_v5_8_controles_lacuna_20261026_v1/README.md)

## Limites e reprodução

O caso 1 demonstra que o Profit preencheu a compra no primeiro candle disponível nesse backtest. Os eventos internos da ordem entre a solicitação e o preenchimento, incluindo o momento em que ficou elegível e a resolução abaixo do segundo, dependem do histórico de ordens do mesmo backtest. O comportamento de alvo, stop e OCO na lacuna depende de `p_Caso=2`, com console, operações, barras, configuração e histórico nativo se disponível. A saída da v5.8 às 10:15 continua classificada como ambígua em OHLC. Nenhuma regra do motor foi alterada. [Protocolo para o caso 2](../WINZ26_v5_8_controles_lacuna_20261026_v1/README.md#execução-manual-no-profit) · [Ambiguidade das 10:15](../WINZ26_v5_8_continuidade_intrabar_20261026_v1/README.md#saída-das-1015-em-2610)

Os quatro originais ficaram no Windows. Cada TXT continha dois blocos de eventos idênticos. O processamento manteve o primeiro bloco, sem reordenar, excluiu cabeçalhos de Debug e normalizou a pontuação da data e de `CurrentBar`. Os oito eventos do caso 0 são avaliações. Dos 11 eventos do caso 1, nove são avaliações e dois são solicitações. O CSV usa CP1252 e tem cinco linhas de preâmbulo, 19 colunas e uma operação. A publicação conserva apenas os campos necessários, sem conta, titular, caminho privado ou preços absolutos. O inventário registra SHA-256 e tamanho dos quatro originais. A verificação do PDF foi feita localmente por extração de texto e inspeção da página renderizada. [Inventário](inventario_fontes.csv)

Foram executados quatro replays locais da sonda sobre as barras privadas preparadas de 26/10, cobrindo casos 0 e 1 com reinício e continuidade. Usou-se a CLI OHLC, 60 s, tick 5, R$ 0,20 por ponto, avaliação pós-preenchimento e rastreamento. O modo continuidade adicionou `--historico-continuo-lacunas`. """Extrai evidência compartilhável da sonda Profit de retomada de lacuna."""

import argparse
import csv
import io
from datetime import datetime
from decimal import Decimal
from hashlib import sha256
from pathlib import Path


def write_csv(path, rows):
    with path.open("x", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(rows[0]), delimiter=";", lineterminator="\n")
        writer.writeheader()
        writer.writerows(rows)


def read_console(data, case):
    blocks = []
    for line in data.decode("utf-8-sig").splitlines():
        if line == "Código em Execução":
            blocks.append([])
        elif line.startswith("LACUNA;"):
            blocks[-1].append(line)
    blocks = [block for block in blocks if block]
    if not blocks or any(block != blocks[0] for block in blocks):
        raise ValueError("O console precisa conter um bloco ou cópias idênticas.")
    result = []
    for index, line in enumerate(blocks[0], 1):
        _, event, *fields = line.split(";")
        values = dict(item.split("=", 1) for item in fields)
        if event == "AVALIACAO":
            if int(values["caso"]) != case:
                raise ValueError("Valor de p_Caso inconsistente no console.")
            context = values
        elif context["seq"] != values["seq"]:
            raise ValueError("Solicitação fora da sequência da avaliação correspondente.")
        stamp = context["hora"].zfill(4)
        result.append({
            "p_caso": case, "ordem_console": index, "evento": event,
            "data_ntsl": context["data"].replace(".", ""), "hora_contexto_console": stamp[:2] + ":" + stamp[2:],
            "currentbar": context["barra"].replace(".", ""), "sequencia_avaliacao": values["seq"],
            "posicao_avaliacao": context["posicao"], "pendentes_avaliacao": context["pendentes"],
        })
    return len(blocks), result


def number(text):
    return Decimal(text.replace(".", "").replace(",", "."))


def read_operation(data):
    def timestamp(value):
        return datetime.strptime(value, "%d/%m/%Y %H:%M:%S")

    raw = csv.reader(io.StringIO(data.decode("cp1252")), delimiter=";")
    header = next(row for row in raw if row[:2] == ["Ativo", "Abertura"])
    values, = [row for row in raw if row]
    operation = dict(zip(header, values, strict=True))
    opening, closing = (timestamp(operation[key]) for key in ("Abertura", "Fechamento"))
    return {
        "p_caso": 1, "pregao": opening.date(), "ativo": operation["Ativo"], "lado": operation["Lado"],
        "quantidade_compra": operation["Qtd Compra"], "quantidade_venda": operation["Qtd Venda"],
        "hora_entrada_csv": opening.time(), "hora_saida_csv": closing.time(),
        "duracao_segundos": int((closing - opening).total_seconds()),
        "variacao_pontos": number(operation["Preço Venda"]) - number(operation["Preço Compra"]),
        "resultado_brl": number(operation["Total"]), "comissoes_brl": number(operation["Comissões"]),
    }


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--originais", type=Path, required=True)
    parser.add_argument("--saida", type=Path, required=True)
    args = parser.parse_args()
    inventory = []
    events = []
    for case in (0, 1):
        file = args.originais / f"console_sondaV3_p_caso{case}.txt"
        content = file.read_bytes()
        copies, rows = read_console(content, case)
        events.extend(rows)
        inventory.append({
            "arquivo": file.name, "bytes": len(content), "sha256": sha256(content).hexdigest(),
            "formato": "TXT UTF-8", "blocos_identicos": copies, "linhas_ou_paginas_retidas": len(rows),
        })
    file = args.originais / "operacoes_sondav3_p_caso1.csv"
    content = file.read_bytes()
    operation = read_operation(content)
    inventory.append({
        "arquivo": file.name, "bytes": len(content), "sha256": sha256(content).hexdigest(),
        "formato": "CSV CP1252", "blocos_identicos": "", "linhas_ou_paginas_retidas": 1,
    })
    file = args.originais / "relatorio_sondav3_p_caso1.pdf"
    content = file.read_bytes()
    inventory.append({
        "arquivo": file.name, "bytes": len(content), "sha256": sha256(content).hexdigest(),
        "formato": "PDF verificado à parte", "blocos_identicos": "", "linhas_ou_paginas_retidas": "",
    })
    args.saida.mkdir(exist_ok=False)
    write_csv(args.saida / "console_profit_normalizado.csv", events)
    write_csv(args.saida / "operacao_profit_relativa.csv", [operation])
    write_csv(args.saida / "inventario_fontes.csv", inventory)


if __name__ == "__main__":
    main()

{ Controle de lacuna. Exclusivo para Backtest OHLC.
  Caso 1: compra solicitada na ultima barra antes da lacuna.
  Caso 2: compra anterior, alvo e stop solicitados antes da lacuna. }
input
  p_Caso(1);
  p_Data(1261026);
  p_InicioLog(927);
  p_EntradaAntecipada(927);
  p_HoraProtecaoInicial(928);
  p_AntesLacuna(929);
  p_Limpeza(941);
  p_FimLog(943);
  p_Alvo(100);
  p_Stop(10000);
  p_FolgaStop(500);

var
  v_Seq, v_Pendentes, v_BarraProtecao : Integer;
  v_EntradaSolicitada, v_LimpezaSolicitada : Boolean;

begin
  if (Date = p_Data) and (Time >= p_InicioLog) and (Time <= p_FimLog) then
  begin
    v_Seq := v_Seq + 1;
    if HasPendingOrders then v_Pendentes := 1
    else v_Pendentes := 0;
    ConsoleLog("LACUNA;AVALIACAO"
               + ";caso=" + FloatToString(p_Caso, 0)
               + ";data=" + FloatToString(Date, 0)
               + ";hora=" + FloatToString(Time, 0)
               + ";barra=" + FloatToString(CurrentBar, 0)
               + ";seq=" + FloatToString(v_Seq, 0)
               + ";posicao=" + FloatToString(PositionQty, 0)
               + ";pendentes=" + FloatToString(v_Pendentes, 0), clYellow);

    if not v_EntradaSolicitada and not HasPosition and not HasPendingOrders
      and (((p_Caso = 1) and (Time = p_AntesLacuna))
        or ((p_Caso = 2) and (Time = p_EntradaAntecipada))) then
    begin
      v_EntradaSolicitada := True;
      ConsoleLog("LACUNA;SOLICITA_COMPRA;seq=" + FloatToString(v_Seq, 0), clYellow);
      BuyAtMarket(1);
    end;

    if (p_Caso = 2) and IsBought and (v_BarraProtecao <> CurrentBar)
      and ((Time = p_HoraProtecaoInicial) or (Time = p_AntesLacuna)) then
    begin
      v_BarraProtecao := CurrentBar;
      ConsoleLog("LACUNA;SOLICITA_ALVO;seq=" + FloatToString(v_Seq, 0)
                 + ";limite=" + FloatToString(BuyPrice + p_Alvo, 0), clYellow);
      SellToCoverLimit(BuyPrice + p_Alvo, 1);
      ConsoleLog("LACUNA;SOLICITA_STOP;seq=" + FloatToString(v_Seq, 0)
                 + ";gatilho=" + FloatToString(BuyPrice - p_Stop, 0)
                 + ";limite=" + FloatToString(BuyPrice - p_Stop - p_FolgaStop, 0), clYellow);
      SellToCoverStop(BuyPrice - p_Stop, BuyPrice - p_Stop - p_FolgaStop, 1);
    end;

    { Limpeza posterior para que a operacao apareca na exportacao. }
    if (Time = p_Limpeza) and not v_LimpezaSolicitada then
    begin
      v_LimpezaSolicitada := True;
      if HasPosition then
      begin
        ConsoleLog("LACUNA;SOLICITA_ZERAGEM_LIMPEZA;seq=" + FloatToString(v_Seq, 0), clYellow);
        ClosePosition;
      end
      else if HasPendingOrders then
      begin
        ConsoleLog("LACUNA;SOLICITA_CANCELAMENTO_LIMPEZA;seq=" + FloatToString(v_Seq, 0), clYellow);
        CancelPendingOrders;
      end;
    end;
  end;
end;
